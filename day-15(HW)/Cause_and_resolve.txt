# Step 1: Make sure you're on main branch
git checkout main

# Step 2: Create conflict.py with one line:
#   message = 'This is from main branch'
git add .
git commit -m "Initial message in main"

# Step 3: Create a new branch 'bug'
git branch bug
git checkout bug

# In conflict.py, CHANGE the same line to:
#   message = 'This is from bug branch'
git add .
git commit -m "Changed message in bug branch"

# Step 4: Go back to main and change the SAME line to:
git checkout main
#   message = 'This is from main branch (edited)'
git add .
git commit -m "Edited message in main branch"

# Step 5: Now try to merge bug into main
git merge bug
# You should get a MERGE CONFLICT!

# Step 6: Resolve by accepting INCOMING changes
# Then:
git add .
